// 箭矢
function playArrow(vol=0.7) {
  const ctx = new AudioContext();
  const buf = ctx.createBuffer(1, ctx.sampleRate*0.1, ctx.sampleRate);
  const d = buf.getChannelData(0);
  for (let i=0; i<d.length; i++) d[i] = (Math.random()*2-1)*Math.exp(-i/(ctx.sampleRate*0.02));
  const s = ctx.createBufferSource();
  const f = ctx.createBiquadFilter();
  const g = ctx.createGain();
  f.type='bandpass'; f.frequency.value=2000; f.Q.value=2;
  s.buffer=buf; s.connect(f); f.connect(g); g.connect(ctx.destination);
  g.gain.setValueAtTime(vol, ctx.currentTime);
  g.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime+0.1);
  s.start();
}
