// 魔法
function playMagic(vol=0.7) {
  const ctx = new AudioContext();
  [0,4,7,12].forEach((semi,i) => {
    const o = ctx.createOscillator();
    const g = ctx.createGain();
    o.connect(g); g.connect(ctx.destination);
    o.type = 'sine';
    const freq = 440 * Math.pow(2, semi/12);
    o.frequency.setValueAtTime(freq*0.5, ctx.currentTime+i*0.04);
    o.frequency.exponentialRampToValueAtTime(freq*2, ctx.currentTime+i*0.04+0.3);
    g.gain.setValueAtTime(0, ctx.currentTime+i*0.04);
    g.gain.linearRampToValueAtTime(vol*0.4, ctx.currentTime+i*0.04+0.05);
    g.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime+i*0.04+0.35);
    o.start(ctx.currentTime+i*0.04);
    o.stop(ctx.currentTime+i*0.04+0.35);
  });
}
